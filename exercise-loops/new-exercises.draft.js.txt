// DRAFT COPY: the coachNote (also spoken as voiceIntro and shown as previewTip),

// Cable rows and pulldowns carry equipment 'Cable', which the Workout Builder
// never deals (GYM_ONLY in fitModeGenerator): they show in the library and the
// info sheet only. Hangs and landmine rows need a bar or a landmine; gearNote
// only names the bar for pull-ups, chin-ups, box jumps, dips and ropes.

const idOf = (muscle, name) => `${muscle.toLowerCase().replace(/\s+/g, '_')}_${name.toLowerCase().replace(/[^a-z0-9]+/g, '_').replace(/^_|_$/g, '')}_core`;

// ex(name, muscle, secondary, equipment, difficulty, movement, cue, carry, opts)
// opts: hold (seconds) for timed work · reps (string) · rest · tempo · style · fixedDose
function ex(name, primaryMuscle, secondaryMuscles, equipment, difficulty, movementType, coachNote, combatCarryover, opts = {}) {
  const hold = opts.hold || null;
  const bodyMapRegion = ['Biceps', 'Triceps'].includes(primaryMuscle) ? 'Arms'
    : ['Quads', 'Hamstrings', 'Calves'].includes(primaryMuscle) ? 'Legs' : primaryMuscle;
  return {
    id: idOf(primaryMuscle, name),
    name,
    primaryMuscle,
    secondaryMuscles,
    equipment,
    difficulty,
    movementType,
    trainingStyle: opts.style || (hold ? 'Endurance' : 'Strength'),
    sets: 3,
    reps: hold ? `${hold}s` : (opts.reps || (equipment === 'Bodyweight' ? '10-15' : '8-12')),
    durationSeconds: hold,
    restSeconds: opts.rest || (hold ? 30 : 60),
    coachNote,
    voiceIntro: `${name}. ${coachNote}`,
    voiceCountingType: hold ? 'timed_hold' : 'rep_count',
    voiceCadenceSeconds: hold ? null : 1.5,
    tempo: opts.tempo || (hold ? null : '2 down, control, strong finish'),
    combatCarryover,
    cardioFinisher: false,
    bodyMapRegion,
    videoUrl: '',
    previewTip: coachNote,
    active: true,
    // The builder keeps this row's own reps and rest (fitModeGenerator).
    ...(opts.fixedDose ? { fixedDose: true } : {}),
  };
}
